#!/usr/bin/env bash
set -euo pipefail
cd "$(dirname "$0")"

echo "[sudo] acquiring credentials..."
sudo -v

# --- One-time repo setup (idempotent)
sudo apt-get update -y
sudo apt-get install -y software-properties-common curl wget gpg
sudo add-apt-repository -y ppa:inkscape.dev/stable || true
wget -qO- https://packages.microsoft.com/keys/microsoft.asc \
    | sudo gpg --dearmor -o /usr/share/keyrings/packages.microsoft.gpg
echo "deb [arch=amd64 signed-by=/usr/share/keyrings/packages.microsoft.gpg] https://packages.microsoft.com/repos/code stable main" \
    | sudo tee /etc/apt/sources.list.d/vscode.list > /dev/null
sudo apt-get update -y

# --- Rust only if needed (cargo.list present)
if [[ -f cargo.list ]] && ! command -v cargo >/dev/null 2>&1; then
    echo "Installing Rust toolchain (rustup) because cargo.list is present..."
    curl -fsSL https://sh.rustup.rs | sh -s -- -y
fi
export PATH="$HOME/.cargo/bin:$PATH"

#+end_src

** install packages from managers

#+begin_src bash :tangle "bootstrap"
# --- Helpers
lines() { grep -vE '^\s*(#|$)' "$1" || true; }	# skip comments/blank lines

apt_installed()   { dpkg-query -W -f='${Status}\n' "$1" 2>/dev/null | grep -q "install ok installed"; }
snap_installed()  { snap list "$1" >/dev/null 2>&1; }
npm_installed()   { npm list -g --depth=0 "$1" >/dev/null 2>&1; }
pip_base_name()   { printf '%s' "$1" | sed 's/[\[=<>].*$//'; }	# strip extras/versions
pip_installed()   { python3 -m pip show "$(pip_base_name "$1")" >/dev/null 2>&1; }
cargo_installed() { cargo install --list 2>/dev/null | awk '{print $1}' | grep -xq "$1"; }
code_installed()  { code --list-extensions 2>/dev/null | grep -xq "$1"; }

ensure_tool() {
    case "$1" in
        code)
            command -v code >/dev/null 2>&1 || sudo apt-get install -y code
            ;;
        npm)
            command -v npm  >/dev/null 2>&1 || sudo apt-get install -y nodejs npm
            ;;
        pip|pip3)
            command -v pip3 >/dev/null 2>&1 || sudo apt-get install -y python3 python3-pip
            ;;
    esac
}

install_pkg() {
    local tool="$1" line="$2"
    local name="${line%% *}"	# first token (handles flags like "--classic")

    case "$tool" in
        apt|apt-get)
            apt_installed "$name" || sudo apt-get install -y $line
            ;;
        snap)
            snap_installed "$name" || sudo snap install $line
            ;;
        npm)
            npm_installed "$name" || sudo npm install -g $line
            ;;
        pip|pip3)
            pip_installed "$line" || python3 -m pip install --user $line
            ;;
        cargo)
            cargo_installed "$name" || cargo install $line
            ;;
        code)
            code_installed "$line" || code --install-extension "$line"
            ;;
        ,*)
            : ;;  # unknown tool: skip
    esac
}

# --- Main loop
shopt -s nullglob
for f in *.list; do
    # skip Gogh theme list; handled in from-source block
    [[ "$(basename "$f")" == "gogh.list" ]] && continue

    tool="${f%.*}"	# e.g., apt.list -> apt
    echo "==> Installing $tool packages..."
    ensure_tool "$tool"

    while IFS= read -r pkg; do
        install_pkg "$tool" "$pkg"
    done < <(lines "$f")

    echo "<== Finished installing $tool packages."
done

#+end_src

** install packages from source

#+begin_src bash :tangle "bootstrap"
# --- From-source installs
echo "==> Installing from-source packages..."

# 1) Starship prompt (install only if missing)
if ! command -v starship >/dev/null 2>&1; then
    curl -sS https://starship.rs/install.sh | sh -s -- -y
fi

# 2) Gogh GNOME Terminal themes driven by gogh.list (install only if missing)
if command -v gnome-terminal >/dev/null 2>&1 && command -v gsettings >/dev/null 2>&1; then
    export TERMINAL=gnome-terminal

    # does a GNOME Terminal profile with visible-name == $1 exist?
    gnome_profile_exists() {
        local want="$1"
        local uuids
        uuids="$(gsettings get org.gnome.Terminal.ProfilesList list 2>/dev/null | tr -d "[],'")" || return 1
        for uuid in $uuids; do
            local vis
            vis="$(gsettings get "org.gnome.Terminal.Legacy.Profile:/org/gnome/terminal/legacy/profiles:/:${uuid}/" visible-name 2>/dev/null || true)"
            vis="${vis//\'}"
            if [[ "$vis" == "$want" ]]; then return 0; fi
        done
        return 1
    }

    # ensure repo present/updated
    mkdir -p "$HOME/src"
    if [[ -d "$HOME/src/gogh/.git" ]]; then
        git -C "$HOME/src/gogh" fetch --all -p >/dev/null 2>&1 || true
        git -C "$HOME/src/gogh" pull --ff-only >/dev/null 2>&1 || true
    else
        git clone https://github.com/Gogh-Co/Gogh.git "$HOME/src/gogh"
    fi

    # permanently normalize default profile name (Gogh expects "Default")
    default_uuid="$(gsettings get org.gnome.Terminal.ProfilesList default | tr -d "'")"
    curr_name="$(gsettings get "org.gnome.Terminal.Legacy.Profile:/org/gnome/terminal/legacy/profiles:/:${default_uuid}/" visible-name | tr -d "'")"
    if [[ "$curr_name" != "Default" ]]; then
        gsettings set "org.gnome.Terminal.Legacy.Profile:/org/gnome/terminal/legacy/profiles:/:${default_uuid}/" visible-name "Default"
    fi

    # read themes from gogh.list (one per line, .sh optional); else default to "atom dracula"
    if [[ -f gogh.list ]]; then
        themes="$(grep -vE '^\s*(#|$)' gogh.list | tr '\n' ' ')"
    fi

    pushd "$HOME/src/gogh/installs" >/dev/null
    for script in $themes; do
        slug="${script%.sh}"
        theme_name="${slug^}"	# "atom" -> "Atom"
        if ! gnome_profile_exists "$theme_name"; then
            "./${slug}.sh" || true	# don’t abort entire bootstrap if a single theme fails
        fi
    done
    popd >/dev/null
fi

echo "<== Finished installing from-source packages."

#+end_src

#+begin_src bash :tangle "bootstrap"
# --- Fonts (group-level logging; idempotent) ---------------------------------
echo "==> Installing fonts..."

# 1) Cascadia Code (prefer apt; skip if installed)
if ! fc-list | grep -qi 'Cascadia Code'; then
	# Try Ubuntu package first (fast and clean)
	if ! sudo apt-get install -y fonts-cascadia-code; then
		# Fallback: download latest TTF zip from GitHub and install to user fonts
		tmp="$(mktemp -d)"
		if curl -fsSL -o "$tmp/cc.zip" "https://github.com/microsoft/cascadia-code/releases/latest/download/CascadiaCode-ttf.zip"; then
			mkdir -p "$HOME/.local/share/fonts/CascadiaCode"
			unzip -o "$tmp/cc.zip" -d "$tmp/cc" >/dev/null
			find "$tmp/cc" -type f -name "*.ttf" -exec cp -f {} "$HOME/.local/share/fonts/CascadiaCode/" \;
			fc-cache -f >/dev/null
		else
			echo "	(warn) could not fetch Cascadia Code from GitHub; skipping"
		fi
		rm -rf "$tmp"
	fi
fi

# 2) Monaspace (clone repo once; copy fonts if missing)
if ! fc-list | grep -qi 'Monaspace'; then
	mkdir -p "$HOME/src" "$HOME/.local/share/fonts/Monaspace"
	if [[ -d "$HOME/src/monaspace/.git" ]]; then
		git -C "$HOME/src/monaspace" pull --ff-only >/dev/null 2>&1 || true
	else
		git clone https://github.com/githubnext/monaspace "$HOME/src/monaspace"
	fi
	# copy any OTF/TTF from repo (covers variable/static)
	find "$HOME/src/monaspace" -type f \( -name "*.otf" -o -name "*.ttf" \) -exec cp -u {} "$HOME/.local/share/fonts/Monaspace/" \;
	fc-cache -f >/dev/null
fi

echo "<== Finished installing fonts."

#+end_src

** ~pdfsearch~: search all PDFs in current directory for string

[[https://stackoverflow.com/a/4643518/11952647][This Stackoverflow answer]] suggests the =find ... -exec sh -c ...=
framework that comprises the body of this short script. [[https://stackoverflow.com/a/4643518/11952647][This answer]]
taught me how to pass the search query to the =-exec= command.
The =-q= flag for =pdftotext= just silences error messages.
The =-m 1= flag for =grep= limits results to one per file.

#+begin_src shell :tangle "/usr/local/bin"
#! /usr/bin/env sh

tosearch=$1
find . -name '*.pdf' -exec sh -c 'pdftotext "{}" -q - | grep -m 1 --with-filename --label="{}" --color "$1"' sh "$tosearch" \;
#+end_src

** [[https://github.com/cheat/cheat][cheat]]: create and view custom interactive cheatsheets on the command line

- Cheatsheets are stored on "cheatpaths", which are directories that contain cheatsheets. Cheatpaths are specified in a YAML config file.
- Can store personal cheatsheets *and* community cheatsheets
- compare [[https://github.com/srsudar/eg][eg]] (in ~pip.list~) and [[https://github.com/tldr-pages/tldr][tldr]] (in ~npm.list~), which *only* provide stock examples
- compare [[https://github.com/gnebbia/kb][kb]], which aims to be a full CLI knowledge base
