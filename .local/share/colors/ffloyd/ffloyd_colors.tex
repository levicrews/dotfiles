% Warm Cream color system
% Categorical, sequential, and diverging palettes designed for #EDE8D9.
% Requires \usepackage{xcolor}; load with \input{ffloyd_colors.tex}.
% All values match the approved palette JSON files and swatches.
% Base background and text colors: https://ffloyd.space/about (light theme).
%
% Basic use (set page and text colors after \begin{document}):
%   \usepackage{xcolor}
%   \input{ffloyd_colors.tex}
%   ...
%   \begin{document}
%   \pagecolor{ff_background}
%   \color{ff_text}
%   Body text with \textcolor{ff_yellow}{yellow emphasis}.
%
% Optional Beamer setup (in the preamble, after loading your theme):
%   \setbeamercolor{normal text}{fg=ff_text,bg=ff_background}
%   \setbeamercolor{structure}{fg=ff_heading}
%   \setbeamercolor{frametitle}{fg=ff_heading,bg=ff_background}
%   \setbeamercolor{alerted text}{fg=ff_yellow}
%   \setbeamercolor{example text}{fg=ff_green}
%
% Optional hyperref setup (after loading hyperref):
%   \hypersetup{colorlinks=true,linkcolor=ff_link,
%               urlcolor=ff_link,citecolor=ff_link}

% Background and default text / supporting colors.
\definecolor{ff_background}{HTML}{EDE8D9} % warm cream page and chart background
\definecolor{ff_text}{HTML}{25606F}       % default body text
\definecolor{ff_heading}{HTML}{164C67}    % headings and titles
\colorlet{ff_link}{ff_heading}           % default links
\definecolor{ff_ink}{HTML}{142C55}        % strongest text emphasis / hover ink
\definecolor{ff_border}{HTML}{9EB29B}     % subtle borders and gridlines
\definecolor{ff_separator}{HTML}{739888}  % stronger rules / blockquote borders
\definecolor{ff_bullet}{HTML}{56867E}     % list markers

% Categorical: unordered groups, in the displayed swatch order.
\definecolor{ff_yellow}{HTML}{CB8A28}     % golden yellow (original highlight)
\definecolor{ff_green}{HTML}{008464}      % jade green (original highlight)
\definecolor{ff_blue}{HTML}{427EAF}       % blue
\definecolor{ff_terracotta}{HTML}{C05B43} % terracotta
\definecolor{ff_violet}{HTML}{7864A0}     % violet
\definecolor{ff_rose}{HTML}{B4567C}       % rose
\definecolor{ff_olive}{HTML}{788238}      % olive
\definecolor{ff_deep_blue}{HTML}{164C67}  % deep blue (original heading color)

% Indexed categorical aliases, for convenient chart styling.
\colorlet{ff_cat_1}{ff_yellow}
\colorlet{ff_cat_2}{ff_green}
\colorlet{ff_cat_3}{ff_blue}
\colorlet{ff_cat_4}{ff_terracotta}
\colorlet{ff_cat_5}{ff_violet}
\colorlet{ff_cat_6}{ff_rose}
\colorlet{ff_cat_7}{ff_olive}
\colorlet{ff_cat_8}{ff_deep_blue}

% Sequential: low to high, with steadily decreasing lightness.
\definecolor{ff_seq_1}{HTML}{EBD495} % pale gold (lowest)
\definecolor{ff_seq_2}{HTML}{C2C16F} % citron
\definecolor{ff_seq_3}{HTML}{86B26E} % leaf
\definecolor{ff_seq_4}{HTML}{3E9F77} % jade
\definecolor{ff_seq_5}{HTML}{1C847C} % teal
\definecolor{ff_seq_6}{HTML}{0F6779} % ocean
\definecolor{ff_seq_7}{HTML}{164C67} % deep blue (highest)

% Diverging: below reference -> reference / zero -> above reference.
% Matching positions on opposite sides have balanced lightness.
% The cream midpoint is distinct from the page background.
\definecolor{ff_div_neg_4}{HTML}{6E3223} % oxide (negative extreme)
\definecolor{ff_div_neg_3}{HTML}{A94C2E} % rust
\definecolor{ff_div_neg_2}{HTML}{CD7F51} % clay
\definecolor{ff_div_neg_1}{HTML}{DDBC95} % sand (near reference)
\definecolor{ff_div_zero}{HTML}{F4F0E4}  % cream (reference / zero)
\definecolor{ff_div_pos_1}{HTML}{A6CBC7} % mist (near reference)
\definecolor{ff_div_pos_2}{HTML}{5EA2A8} % seafoam
\definecolor{ff_div_pos_3}{HTML}{21778D} % teal
\definecolor{ff_div_pos_4}{HTML}{164C67} % deep blue (positive extreme)
